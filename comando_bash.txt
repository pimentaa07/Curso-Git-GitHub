-pwd (caminho absoluto de onde estou)
- mkdir (cria pasta)
- cat (mostrar conteúdo do arquivo)
-git init . (CRIAR O GIT)
-rm -rf .git/ (apaga  o git)
-git status
-git add nome_arquivo
-git commit -m "" (descrição do save)
-git config --global user.email "gustavo.l.pimenta@ufv.br"
-git add nome arquivo(atualizar)
-git log (ver todos os saves)
